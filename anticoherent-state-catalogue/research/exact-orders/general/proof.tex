\title{An explicit construction of pure symmetric states of exactly anticoherence order $t$}
\author{Catalogue research note}
\date{5 November 2026}
\begin{document}\maketitle
The support-gap and moment-matching approach is established in Baguette, Damanet, Giraud and Martin, Phys. Rev. A 92, 052333 (2015), Secs. V--VI. The construction below adds a rational formula for weights on a complete arithmetic grid, an explicit sufficient size bound, and a strictly positive next-order coherence certifying exact order. No claim of an optimal threshold or priority over all existing constructions is made.
\begin{theorem}
For every integer $t\ge1$ and every integer $N\ge4t(t+1)^4$, there is an explicitly specified normalized pure symmetric $N$-qubit state which is $t$-AC and not $(t+1)$-AC. Its Dicke coefficients are square roots of positive rational numbers.
\end{theorem}
Put $s=t+1$, $m=\lfloor N/s\rfloor$, $R=N-sm$, $x_i=si/N$, and
\[
q_i=\begin{cases}s/(N+1),&0\le i<m,\\(R+1)/(N+1),&i=m.\end{cases}
\]
Let $v(x)=(1,x,\ldots,x^t)^T$ and define
\[
G=\sum_{i=0}^m q_i v(x_i)v(x_i)^T,\qquad
\mu=\frac1{N+1}\sum_{r=0}^N v(r/N),\qquad
w_i=q_i v(x_i)^TG^{-1}\mu.
\]
The state is
\[
\lvert\psi_{N,t}\rangle=\sum_{i=0}^m\sqrt{w_i}\,\D{N}{si}.
\]
All matrices in this formula have size $t+1$, and all their entries are rational. The sums can be evaluated by Faulhaber's formulas. The matrix $G$ is positive definite whenever $m\ge t$, because a degree-at-most-$t$ polynomial cannot vanish at $t+1$ distinct nodes unless it is zero.

\paragraph{Moment matching.}
Directly from the definition,
\[
\sum_i w_i v(x_i)=GG^{-1}\mu=\mu.
\]
Thus $\sum_iw_i=1$ and $\sum_iw_i(si)^a=(N+1)^{-1}\sum_{r=0}^Nr^a$ for $0\le a\le t$.

\paragraph{Positivity for every sufficiently large $N$.}
The weight formula is invariant under any invertible change of polynomial basis. Use
\[
f_\ell(x)=\sqrt{2\ell+1}\,P_\ell(2x-1),\quad0\le\ell\le t,
\]
where $P_\ell$ is the Legendre polynomial. This basis is orthonormal for uniform measure on $[0,1]$, and $f_0=1$. The bounds $|P_\ell|\le1$ and $|P'_\ell|\le\ell(\ell+1)/2$ give
\[
\sup_x\|f(x)\|\le t+1=:M,\qquad
\sup_x\|f'(x)\|\le t(t+1)^2=:D.
\]
For $U$ uniform on $[0,1)$ put $Z=\lfloor(N+1)U\rfloor/N$ and $X=s\lfloor NZ/s\rfloor/N$. Then $Z$ is uniform on the Dicke grid, $\Pr(X=x_i)=q_i$, and
\[
|U-Z|\le\frac1{N+1},\quad |Z-X|\le\frac tN,\quad |U-X|\le\frac{t+1}N.
\]
Write $H=\mathbb E[f(X)f(X)^T]$ and $b=\mathbb E[f(Z)]-\mathbb E[f(X)]$. The derivative of $ff^T$ has operator norm at most $2MD$, so
\[
\|H-I\|\le\frac{2t(t+1)^4}{N}\le\frac12,\qquad
\|b\|\le\frac{t^2(t+1)^2}{N}.
\]
Since $He_0=\mathbb E f(X)$,
\[
\frac{w_i}{q_i}=1+f(x_i)^TH^{-1}b
\ge1-\frac{2t^2(t+1)^3}{N}
\ge1-\frac{t}{2(t+1)}>\frac12.
\]
Thus every $w_i$ is strictly positive.

\paragraph{Anticoherence through order $t$.}
Distinct occupied excitation indices differ by at least $s=t+1$, so all off-diagonal entries of every reduction through order $t$ vanish. The diagonal entries of the $q$-body reduction are
\[
(\rho_q)_{aa}=\sum_r |c_r|^2
\frac{\binom ra\binom{N-r}{q-a}}{\binom Nq}.
\]
For fixed $q\le t$, the numerator is a degree-at-most-$q$ polynomial in $r$. Moment matching makes this equal to the uniform average over $r=0,\ldots,N$. Vandermonde's identity gives $(\rho_q)_{aa}=1/(q+1)$. Hence $\rho_q=I_{q+1}/(q+1)$.

\paragraph{Failure at the next order.}
Because every occupied coefficient is positive real,
\[
(\rho_s)_{0s}=\sum_{i=0}^{m-1}\sqrt{w_iw_{i+1}}
\frac{\binom{N-s}{si}}{\sqrt{\binom N{si}\binom N{s(i+1)}}}>0.
\]
Thus $\rho_s$ is not maximally mixed, proving exact AC order $t$. The same construction works below the stated threshold whenever $m\ge t$ and all computed weights are positive. The threshold is sufficient and deliberately conservative; it does not characterize existence for every small pair $(N,t)$.

\begin{proposition}
For every $N\ge7$, set $k=\lfloor(N-1)/2\rfloor$ and
\[
p_0=\frac{3k-N+1}{6k},\qquad
p_k=\frac{N(N-1)}{6k(N-k)},\qquad
p_N=\frac{2N+1-3k}{6(N-k)}.
\]
Then $\sqrt{p_0}\D{N}{0}+\sqrt{p_k}\D{N}{k}+\sqrt{p_N}\D{N}{N}$ has exactly AC order two.
\end{proposition}
Indeed, all weights are positive and sum to one, and $k,N-k\ge3$. Their first two raw excitation moments are $N/2$ and $N(2N+1)/6$, matching the uniform Dicke distribution. The third raw moment differs by
\[
\sum_rp_rr^3-\frac{N^2(N+1)}4=\frac{N(N-1)(N-2k)}{12}>0.
\]
This proves failure of third-order anticoherence. The $N=6$ example is $(\sqrt2\D6{0}+\sqrt5\D6{3}+\sqrt2\D6{6})/3$, for which $(\rho_3)_{03}=\sqrt2/9$. At $t=1$, GHZ states give exact order one for every $N\ge2$.
